% --- Automata in robotic systems ---
\begin{frame}{Automata in robotic systems}
  \justifying
  Beyond the lower layers of perception, planning, and control, every modern \textbg{autonomous system} needs an \textbg{orchestration layer}: the top of the stack where everything has become a \textbg{software abstraction}, \textbg{user input} can be parsed, and the \textbg{mission} is \textbg{executed}.\\
  \bigskip
  We present \textbg{three families} of possible \textbg{implementations} for this layer.
  \begin{itemize}
    \item \textbg{Finite-state machines}: the traditional starting point.
    \item \textbg{Behavior trees}: when FSMs become unwieldy.
    \item \textbg{LLM-based agents}: a glimpse of where the field is heading.
  \end{itemize}
  \bigskip
  \begin{block}{}
    \centering
    We use the term \textbf{automata} for this layer.\\
    This is unrelated to the formal-language automata seen in other courses.
  \end{block}
\end{frame}

% --- Finite-state machines ---
\begin{frame}{Finite-state machines}{Definition}
  \justifying
  An FSM is a \textbg{mathematical model of computation}.
  \begin{block}{Formal definition}
    \justifying
    A finite-state machine is a tuple $(S, \Sigma, \delta, s_0, F)$, where
    \begin{itemize}
      \item $S$ is a finite set of \textbf{states};
      \item $\Sigma$ is a finite set of \textbf{input symbols} (events, triggers);
      \item $\delta : S \times \Sigma \to S$ is the \textbf{transition function};
      \item $s_0 \in S$ is the \textbf{initial state};
      \item $F \subseteq S$ is the set of \textbf{terminal} (or \emph{accepting}) \textbf{states}.
    \end{itemize}
  \end{block}
  At any time, the machine is in \textbg{exactly one state} from $S$; an incoming event \textbg{triggers a transition} according to $\delta$.
\end{frame}
\begin{frame}{Finite-state machines}{The transition table}
  \justifying
  $\delta$ must be \textbg{well-defined}: every (state, trigger) pair must have at most one outgoing transition, otherwise the machine is non-deterministic and a different formalism is needed.\\
  \bigskip
  In practice, $\delta$ is encoded as a \textbg{table}: each \textbg{row} associates a \textbg{source state}, a \textbg{trigger}, and a \textbg{destination state}.\\
  \bigskip
  This is the form FSMs take in code: a \textbg{list} of triples, easy to read and easy to extend.
\end{frame}
\begin{frame}{Finite-state machines}{Diagrammatic representation}
  \justifying
  FSMs are usually drawn as \textbg{directed graphs}: circles are states, arrows are transitions labeled with the triggering event. An incoming arrow with no source marks the \textbg{initial state}; a double circle marks a \textbg{terminal state}.

% --- Passive and active FSMs ---
\begin{frame}{Passive and active FSMs}{Passive}
  \justifying
  A \textbg{passive} FSM is a \textbg{data structure}: it \textbg{tracks the state of something} (a component, a process, a robot), and is updated by external code.\\
  \bigskip
  \textbg{The FSM itself does not run anything.} It answers questions like the following.
  \begin{itemize}
    \item \textbg{Where am I?} (Current state)
    \item \textbg{Can I do X?} (Is the current state allowed to transition on trigger X?)
  \end{itemize}
  \bigskip
  \begin{block}{}
    \centering
    \textbf{Passive FSMs are observers and reporters, not executors.}
  \end{block}
\end{frame}
\begin{frame}{Passive and active FSMs}{Active}
  \justifying
  An \textbg{active} FSM \textbg{embeds execution logic}: a routine is associated to each state, and is \textbg{executed when the state is entered}.\\
  \bigskip
  The routine's \textbg{return value} may be the \textbg{trigger} for the next transition.\\
  The machine \textbg{progresses autonomously} until a \textbg{terminal state} is reached.\\
  \bigskip
  \begin{block}{}
    \centering
    \textbf{Active FSMs are full executors of a mission or process.}
  \end{block}
\end{frame}

% --- Active FSMs ---
\begin{frame}{Active FSMs}{Execution model}
  \justifying
  Each step of an \textbg{active FSM} follows a \textbg{fixed cycle}:
  \begin{enumerate}
    \item the machine is in a \textbg{state} $s \in S$ with \textbg{associated routine} $R_s$;
    \item $R_s$ is invoked and \textbg{runs to completion};
    \item $R_s$ \textbg{returns a trigger} $t \in \Sigma$;
    \item the machine looks up $\delta(s, t) = s'$ and \textbg{transitions} to $s'$;
    \item if $s' \in F$ (\emph{i.e.}, it is \textbg{terminal}), the \textbg{machine halts}; otherwise the cycle \textbg{repeats}.
  \end{enumerate}
  \bigskip
  \begin{block}{}
    \centering
    \textbf{An FSM run is a sequence of routine calls, threaded together by the transition table.}
  \end{block}
\end{frame}
\begin{frame}{Active FSMs}{Implications}
  \justifying
  This execution model has direct consequences:
  \begin{itemize}
    \item \textbg{every routine must terminate}, otherwise the machine is stuck;
    \item \textbg{progress is strictly sequential}: one state, one routine, at a time;
    \item there is \textbg{no preemption}: while $R_s$ runs, no external event can interrupt it.
  \end{itemize}
  \bigskip
  \begin{alertblock}{Consequence}
    \justifying
    \textbr{A routine that must react to multiple conditions has to monitor them explicitly inside its body}: there is no FSM-level mechanism for reacting to events occurring during state execution.
  \end{alertblock}
\end{frame}

% --- transitions_ros ---
\begin{frame}{\texttt{transitions\_ros}}
  \justifying
  Our reference implementation is the \href{https://github.com/dotX-Automation/transitions_ros}{\color{blue}\underline{\texttt{transitions\_ros}}} Python library, built on top of the \href{https://github.com/pytransitions/transitions}{\color{blue}\underline{\texttt{transitions}}} Python package.\\
  \bigskip
  Where \texttt{transitions} focuses on FSMs as \textbg{state trackers} attached to arbitrary objects,\\\texttt{transitions\_ros} focuses on the FSM as a \textbg{controller}, and adds
  \begin{itemize}
    \item \textbg{full ROS 2 integration}: the machine owns a ROS node and routines can use it to communicate;
    \item explicit support for the \textbg{active} execution model described before;
    \item a \textbg{generic executor node} that \textbg{loads an FSM definition at runtime}.
  \end{itemize}
\end{frame}
\begin{frame}{\texttt{transitions\_ros}}{Architecture}
  \justifying
  The library is built around \textbg{two ideas}:
  \begin{itemize}
    \item a \texttt{Machine} class extending \texttt{transitions.Machine}, that optionally \textbg{holds a ROS 2 node} and \textbg{supports active routines};
    \item a \textbg{generic ROS 2 node} (\texttt{machine\_executor}) that \textbg{loads an FSM definition from parameters and a Python module}, then \textbg{runs it}.
  \end{itemize}
  \bigskip
  \begin{block}{The executor pattern}
    \justifying
    The \textbf{\texttt{machine\_executor}} is \textbf{mission-agnostic}: it knows nothing about the specific FSM it runs.\\
    The FSM definition lives in a \textbf{separate Python module loaded at machine instantiation}.\\
    \textbf{Reconfiguring the mission means swapping the module}, not rebuilding the executor.
  \end{block}
\end{frame}
\begin{frame}[fragile]{\texttt{transitions\_ros}}{Defining the transitions table}
  \begin{columns}
    \column{.9\textwidth}
    \begin{lstlisting}[language=Python, caption=Example of transitions table in \texttt{transitions\_ros}.]
transitions_table = [
    ['START',        'INIT',          'WAIT_START'],
    ['READY',        'WAIT_START',    'FIRST_TAKEOFF'],
    ['AIRBORNE',     'FIRST_TAKEOFF', 'EXPLORE'],
    # ...
    ['MISSION_DONE', 'DONE',          'RTH'],
    ['AT_HOME',      'RTH',           'COMPLETED']
]
\end{lstlisting}
  \end{columns}
  \bigskip
  Each \textbg{row} encodes one entry of $\delta$ as \texttt{[trigger, source, destination]}.\\
  \textbg{Trigger strings are what state routines return} when they want to \textbg{drive the corresponding transition}.
\end{frame}
\begin{frame}[fragile]{\texttt{transitions\_ros}}{Defining the states and their routines}
  \begin{columns}
    \column{.9\textwidth}
    \begin{lstlisting}[language=Python, caption=A state routine and a states table.]
def init_routine(node: MachineExecutorNode,
                 wait_servers: bool,
                 timeout_sec: float) -> str:
    """Wait for the mission start command."""
    node.get_logger().warn("SYSTEMS ONLINE")
    # ... do something, possibly use the ROS node ...
    return 'START'

states_table = [
    {'name': 'INIT',      'routine': init_routine},
    {'name': 'RTH',       'routine': rth_routine},
    {'name': 'COMPLETED', 'routine': completed_routine}
]
\end{lstlisting}
  \end{columns}
\end{frame}
\begin{frame}{\texttt{transitions\_ros}}{Runtime considerations}
  \justifying
  Once started, the \textbg{executor node}:
  \begin{itemize}
    \item \textbg{iterates the active machine} via \texttt{run()}, calling each routine in turn;
    \item \textbg{exposes the ROS node to the routines}, which can publish, subscribe, call services and actions, and use parameters as any ROS 2 client would;
    \item \textbg{can be stopped, reconfigured, and restarted} at runtime without rebuilding.
  \end{itemize}
  \bigskip
  \begin{block}{}
    \centering
    \textbf{The mission is just Python code that uses ROS and returns the right trigger strings.}
  \end{block}
  \bigskip
  \begin{alertblock}{}
    \centering
    \textbr{Such an orchestrator node is usually a client for ROS~2 servers running on the robot.}
  \end{alertblock}
\end{frame}

\begin{frame}{Use case: Leonardo Drone Contest}{Pain points of LDC23-LDC24}
  Mission was still linear(-ish) in LDC22, \textbg{things started to get wild from 2023 onwards}:
  \begin{itemize}
    \item \textbg{two agents}, a UAV and UGV, sharing a common network;
    \item \textbg{Human-Machine Interfaces (HMI)} as a requirement;
    \item \textbg{many more tasks}, some shared between UAV and UGV;
    \item \textbg{many more asynchronous events} with external triggers;
    \item \textbg{external software to interface} for mission generation and management.
  \end{itemize}
  \bigskip
  \begin{alertblock}{}
    \centering
    \textbr{We started to hit the limits of an FSM.}
  \end{alertblock}

% --- When FSMs stop scaling ---
\begin{frame}{When FSMs stop scaling}{The qualitative story}
  \justifying
  As soon as the mission becomes \textbg{reactive} --- the robot must continuously \textbg{monitor multiple conditions} --- \textbg{FSMs start to show their limitations}:
  \begin{itemize}
    \item \textbg{every monitored condition must be checked inside every routine}, else the event is missed;
    \item \textbg{adding a new condition often means touching many transitions} at once;
    \item \textbg{recovery and error-handling paths quickly outnumber the nominal ones}.
  \end{itemize}
  \bigskip
  The FSM stops being a clean model of the mission and becomes a \textbg{tangle of spaghetti code}.
\end{frame}
\begin{frame}{When FSMs stop scaling}{The quantitative story}
  \justifying
  If the robot must independently track $N$ binary conditions, the number of distinct \textbg{combinations of conditions} is $2^N$.\\
  \bigskip
  In the worst case, faithfully modeling the mission requires \textbg{one state per combination}: the state space grows \textbg{exponentially}.
  \begin{alertblock}{State explosion}
    \centering
    \textbr{$N = 5$ conditions $\to$ up to $32$ states.}\\
    \textbr{$N = 10$ conditions $\to$ up to $1024$ states.}
  \end{alertblock}
  \bigskip
  Even with disciplined modeling, the resulting machine is hard to read, hard to test, and hard to evolve.
\end{frame}
\begin{frame}{When FSMs stop scaling}{Hierarchical FSMs and statecharts}
  \justifying
  The classical mitigation is the \textbg{hierarchical FSM}, or \href{https://www.state-machine.com/doc/Harel87.pdf}{\color{blue}\underline{\textbf{Harel statechart}}}: \textbg{states may contain nested sub-states}, with parallel regions and history points.\\
  \bigskip
  This is what one finds in \textbg{industrial tools} (UML state machines, MATLAB Stateflow) and it does tame explosion, but \textbg{the underlying model is still state-centric}:
  \begin{itemize}
    \item \textbg{reactivity must still be encoded as transitions};
    \item \textbg{modularity is limited to nesting};
    \item \textbg{composition of independent behaviors remains limited} by structural requirements.
  \end{itemize}
  \bigskip
  A different idiom is needed.
\end{frame}
